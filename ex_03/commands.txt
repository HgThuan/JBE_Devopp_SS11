# ==============================================================================
# DANH SÁCH LỆNH TẠO & CẤU HÌNH BẢO MẬT SSH KEY - BÀI TẬP 3 (SS11)
# ==============================================================================

# --- PHẦN 1: TRÊN MÁY CÁ NHÂN (macOS) ---
# 1. Tạo cặp SSH Key (Private & Public) chuẩn Ed25519
ssh-keygen -t ed25519 -C "hoangvanthuan010@gmail.com"

# 2. Copy Public Key lên VPS vào ~/.ssh/authorized_keys của user 'deploy'
ssh-copy-id -i ~/.ssh/id_ed25519.pub deploy@163.223.9.193
# (Hoặc thiết lập trực tiếp trên VPS với quyền 700 cho ~/.ssh và 600 cho authorized_keys)

# 3. Kiểm tra đăng nhập bằng SSH Key cho user 'deploy' TRƯỚC khi tắt root/password
ssh -i ~/.ssh/id_ed25519 deploy@163.223.9.193 "whoami && sudo whoami"

# --- PHẦN 2: TRÊN VPS (với quyền sudo) ---
# 4. Sao lưu cấu hình SSH
sudo cp /etc/ssh/sshd_config /etc/ssh/sshd_config.bak

# 5. Vô hiệu hóa đăng nhập root trực tiếp và vô hiệu hóa đăng nhập bằng mật khẩu
sudo sed -i -E 's/^#?PermitRootLogin.*/PermitRootLogin no/' /etc/ssh/sshd_config
sudo sed -i -E 's/^#?PasswordAuthentication.*/PasswordAuthentication no/' /etc/ssh/sshd_config
sudo sed -i -E 's/^#?PubkeyAuthentication.*/PubkeyAuthentication yes/' /etc/ssh/sshd_config

# 6. Kiểm tra cú pháp cấu hình và khởi động lại SSH Daemon
sudo sshd -t
sudo systemctl restart ssh

# --- PHẦN 3: KIỂM THỬ TỪ MÁY CÁ NHÂN ---
# 7. Đăng nhập bằng root -> Phải bị từ chối: Permission denied (publickey)
ssh root@163.223.9.193

# 8. Đăng nhập bằng user deploy -> Thành công không cần mật khẩu
ssh deploy@163.223.9.193
